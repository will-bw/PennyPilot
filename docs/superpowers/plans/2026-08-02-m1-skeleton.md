> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 搭建 PennyPilot Android 项目骨架，实现手动记账、流水列表、分类管理的完整闭环。

**Architecture:** 单模块 Clean 架构（data/domain/ui 分包），Room 持久化 + Hilt DI + Compose Navigation。M1 只含手动记账流，AI/通知/洞察在后续里程碑叠加。

// app/src/main/java/com/pennypilot/app/util/MoneyUtil.kt
package com.pennypilot.app.util

import java.text.DecimalFormat
import java.text.DecimalFormatSymbols
import java.util.Locale

object MoneyUtil {
    private val symbols = DecimalFormatSymbols(Locale.US)
    private val yuanFormat = DecimalFormat("#,##0.00", symbols)

    /** 流水列表显示：支出 "−18.00"（墨色），收入 "+8,000.00"（暖红） */
    fun formatForDisplay(amountCents: Long, isExpense: Boolean): String {
        if (amountCents == 0L) return "0.00"
        val absYuan = kotlin.math.abs(amountCents) / 100.0
        val formatted = yuanFormat.format(absYuan)
        return if (isExpense) "−$formatted" else "+$formatted"
    }

    /** Hero 大数字：无符号，"3,286.50" */
    fun formatHeroAmount(amountCents: Long): String {
        return yuanFormat.format(amountCents / 100.0)
    }

    /** 编辑框：分 → 纯数字字符串 "18.00" */
    fun centsToYuanString(amountCents: Long): String {
        return yuanFormat.format(kotlin.math.abs(amountCents) / 100.0)
    }

    /** 用户输入元 → 分。无效输入返回 0 */
    fun parseYuanToCents(input: String): Long {
        return try {
            val yuan = input.trim().toDouble()
            (yuan * 100).toLong()
        } catch (_: NumberFormatException) {
            0L
        }
    }
}

// data/db/dao/TransactionDao.kt
package com.pennypilot.app.data.db.dao

import androidx.room.*
import com.pennypilot.app.data.db.entity.TransactionEntity
import com.pennypilot.app.domain.model.TxStatus
import kotlinx.coroutines.flow.Flow

@Dao
interface TransactionDao {
    @Insert
    suspend fun insert(tx: TransactionEntity): Long

    @Update
    suspend fun update(tx: TransactionEntity)

    @Query("UPDATE transactions SET status = :status WHERE id = :id")
    suspend fun updateStatus(id: Long, status: TxStatus)

    @Query("SELECT * FROM transactions WHERE status != 'IGNORED' AND occurred_at BETWEEN :start AND :end ORDER BY occurred_at DESC")
    fun observeByMonth(start: Long, end: Long): Flow<List<TransactionEntity>>

    @Query("SELECT * FROM transactions WHERE id = :id")
    suspend fun getById(id: Long): TransactionEntity?

import com.pennypilot.app.data.db.dao.TransactionDao
import com.pennypilot.app.data.db.entity.TransactionEntity
import com.pennypilot.app.domain.model.TxStatus
import dagger.hilt.android.lifecycle.HiltViewModel
import kotlinx.coroutines.ExperimentalCoroutinesApi
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch
import java.time.LocalDate
import java.time.ZoneId
import javax.inject.Inject

data class DayGroup(
    val label: String,       // "今天" / "昨天" / "8月1日 · 周六"
    val netAmount: Long,     // 日结余（显示用）
    val transactions: List<TransactionEntity>
)

data class TransactionsUiState(
    val monthExpense: Long = 0,
    val monthIncome: Long = 0,
    val pendingCount: Int = 0,
    val dayGroups: List<DayGroup> = emptyList(),
    val selectedMonth: LocalDate = LocalDate.now()
)

@HiltViewModel
class TransactionsViewModel @Inject constructor(
    private val txDao: TransactionDao
) : ViewModel() {

    private val _selectedMonth = MutableStateFlow(LocalDate.now())

    @OptIn(ExperimentalCoroutinesApi::class)
    val uiState = _selectedMonth.flatMapLatest { month ->
        val start = month.withDayOfMonth(1).atStartOfDay(ZoneId.systemDefault()).toInstant().toEpochMilli()
        val end = month.plusMonths(1).withDayOfMonth(1).atStartOfDay(ZoneId.systemDefault()).toInstant().toEpochMilli()

        combine(
            txDao.observeMonthExpense(start, end),
            txDao.observeMonthIncome(start, end),
            txDao.observePendingCount(),
            txDao.observeByMonth(start, end)
        ) { expense, income, pending, txs ->
            TransactionsUiState(
                monthExpense = expense,
                monthIncome = -income, // income 存负数，取反
                pendingCount = pending,
                dayGroups = groupByDay(txs),
                selectedMonth = month
            )
        }
    }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), TransactionsUiState())

    fun shiftMonth(delta: Int) {
        _selectedMonth.update { it.plusMonths(delta.toLong()) }
    }

    fun confirmTx(id: Long) = viewModelScope.launch {
        txDao.updateStatus(id, TxStatus.CONFIRMED)
    }

    fun ignoreTx(id: Long) = viewModelScope.launch {
        txDao.updateStatus(id, TxStatus.IGNORED)
    }

    private fun groupByDay(txs: List<TransactionEntity>): List<DayGroup> {
        val today = LocalDate.now()
        return txs.groupBy {
            java.time.Instant.ofEpochMilli(it.occurredAt).atZone(ZoneId.systemDefault()).toLocalDate()
        }.toSortedMap(reverseOrder()).map { (date, dayTxs) ->
            val label = when (date) {
                today -> "今天"
                today.minusDays(1) -> "昨天"
                else -> "${date.monthValue}月${date.dayOfMonth}日"
            }
            DayGroup(label = label, netAmount = -dayTxs.sumOf { it.amount }, transactions = dayTxs)
        }
    }
}

package com.pennypilot.app.ui.edit

import androidx.lifecycle.SavedStateHandle
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.pennypilot.app.data.db.dao.CategoryDao
import com.pennypilot.app.data.db.dao.TransactionDao
import com.pennypilot.app.data.db.entity.CategoryEntity
import com.pennypilot.app.data.db.entity.TransactionEntity
import com.pennypilot.app.domain.model.TxSource
import com.pennypilot.app.domain.model.TxStatus
import com.pennypilot.app.domain.model.TxType
import dagger.hilt.android.lifecycle.HiltViewModel
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch
import javax.inject.Inject

data class EditUiState(
    val amountText: String = "",
    val type: TxType = TxType.EXPENSE,
    val selectedCategoryId: Long = 0,
    val note: String = "",
    val merchant: String = "",
    val accountTag: String = "",
    val categories: List<CategoryEntity> = emptyList(),
    val isEditing: Boolean = false
)

@HiltViewModel
class EditTransactionViewModel @Inject constructor(
    private val txDao: TransactionDao,
    private val catDao: CategoryDao,
    savedStateHandle: SavedStateHandle
) : ViewModel() {

    private val txId: Long = savedStateHandle.get<Long>("id") ?: 0L

    private val _state = MutableStateFlow(EditUiState())
    val state: StateFlow<EditUiState> = _state.asStateFlow()

    init {
        viewModelScope.launch {
            catDao.observeByType(TxType.EXPENSE).collect { cats ->
                _state.update { it.copy(categories = cats, selectedCategoryId = it.selectedCategoryId ?: cats.firstOrNull()?.id ?: 0) }
            }
        }
        if (txId > 0L) {
            viewModelScope.launch {
                txDao.getById(txId)?.let { tx ->
                    _state.update {
                        it.copy(
                            amountText = com.pennypilot.app.util.MoneyUtil.centsToYuanString(tx.amount),
                            type = tx.type,
                            selectedCategoryId = tx.categoryId,
                            note = tx.note ?: "",
                            merchant = tx.merchant ?: "",
                            accountTag = tx.accountTag ?: "",
                            isEditing = true
                        )
                    }
                }
            }
        }
    }

    fun updateAmount(v: String) { _state.update { it.copy(amountText = v) } }
    fun updateNote(v: String) { _state.update { it.copy(note = v) } }
    fun updateMerchant(v: String) { _state.update { it.copy(merchant = v) } }
    fun selectCategory(id: Long) { _state.update { it.copy(selectedCategoryId = id) } }

    fun save(onDone: () -> Unit) {
        val s = _state.value
        val cents = com.pennypilot.app.util.MoneyUtil.parseYuanToCents(s.amountText)
        if (cents == 0L) return

        viewModelScope.launch {
            val amount = if (s.type == TxType.EXPENSE) cents else -cents
            if (s.isEditing) {
                txDao.update(TransactionEntity(
                    id = txId, amount = amount, type = s.type,
                    categoryId = s.selectedCategoryId, note = s.note.ifBlank { null },
                    merchant = s.merchant.ifBlank { null }, accountTag = s.accountTag.ifBlank { null },
                    occurredAt = System.currentTimeMillis(), source = TxSource.MANUAL, status = TxStatus.CONFIRMED
                ))
            } else {
                txDao.insert(TransactionEntity(
                    amount = amount, type = s.type,
                    categoryId = s.selectedCategoryId, note = s.note.ifBlank { null },
                    merchant = s.merchant.ifBlank { null }, accountTag = s.accountTag.ifBlank { null },
                    occurredAt = System.currentTimeMillis(), source = TxSource.MANUAL, status = TxStatus.CONFIRMED
                ))
            }
            onDone()
        }
    }

    fun delete(onDone: () -> Unit) {
        viewModelScope.launch {
            if (txId > 0) txDao.deleteById(txId)
            onDone()
        }
    }
}

package com.pennypilot.app.ui.categories

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.pennypilot.app.data.db.dao.CategoryDao
import com.pennypilot.app.data.db.entity.CategoryEntity
import com.pennypilot.app.domain.model.CategoryColors
import com.pennypilot.app.domain.model.TxType
import dagger.hilt.android.lifecycle.HiltViewModel
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch
import javax.inject.Inject

@HiltViewModel
class CategoriesViewModel @Inject constructor(
    private val catDao: CategoryDao
) : ViewModel() {

    val categories: StateFlow<List<CategoryEntity>> = catDao.observeAll()
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())

    fun addCategory(name: String, icon: String, type: TxType) {
        viewModelScope.launch {
            val maxIdx = catDao.maxColorIndex()
            val maxOrder = categories.value.maxOfOrNull { it.sortOrder } ?: -1
            catDao.insert(CategoryEntity(
                name = name, icon = icon,
                colorIndex = CategoryColors.nextIndex(maxIdx),
                type = type, sortOrder = maxOrder + 1, isPreset = false
            ))
        }
    }

    fun toggleEnabled(cat: CategoryEntity) {
        viewModelScope.launch {
            catDao.update(cat.copy(enabled = !cat.enabled))
        }
    }
}
